\vspace{1em}
where $\d(f)$ is the Discrete Fourier transform (DFT) at frequency $f$ and
\begin{align*}
    \d(f) &\sim {\cal N}_p(\0, \S(f)) \\
    \I(f) = \d(f)\d(f)^* &\sim {\cal CW}(\S(f), 1) \\
     \Y(f) =  N_b \bar{\I}(f) = \sum_{i=1}^{N_b}\I^{(i)}(f) &\sim {\cal CW}(\S(f), N_b) \\
     \bar{\Y}_h & \dot{\sim} {\cal CW}(\S(\bar{f}_h), N_b \times N_h)
\end{align*}

Thus, the approximated likelihood for $\bar{\Y}$ is given by
\begin{align*}
\mathcal{L}(\bar{\Y}|\S) \propto \prod_{h=1}^{N_c} \left|\S(\bar{f}_h)\right|^{-N_b \times N_h} \exp\left(- \tr\left[ \S(\bar{f}_h)^{-1} \bar{\Y}_h\right]\right)    
\end{align*}

	\vspace{1em}
	where $\d(f)$ is the Discrete Fourier transform (DFT) at frequency $f$ and
	\begin{align*}
		\d(f)                                                   & \sim{\cal N}_{p}(\0, \S(f))                              \\
		\I(f) = \d(f)\d(f)^{*}                                  & \sim{\cal CW}(\S(f), 1)                                  \\
		\Y(f) = N_{b} \bar{\I}(f) = \sum_{i=1}^{N_b}\I^{(i)}(f) & \sim{\cal CW}(\S(f), N_{b})                              \\
		\bar{\Y}_{h}                                            & \dot{\sim}{\cal CW}(\S(\bar{f}_{h}), N_{b} \times N_{h})
	\end{align*}

	Thus, the approximated likelihood for $\bar{\Y}$ is given by
	\begin{align*}
		\mathcal{L}(\bar{\Y}|\S) \propto \prod_{h=1}^{N_c}\left|\S(\bar{f}_{h})\right|^{-N_b \times N_h}\exp\left(- \tr\left[ \S(\bar{f}_{h})^{-1}\bar{\Y}_{h}\right]\right)
	\end{align*}

\mathcal{L}_{\mathrm{blocked}}(\S\mid\Y)
&\propto \prod_{k=1}^{N_l}\left|\S(f_k)\right|^{-N_b}
\exp\!\left(-\frac{1}{T_b}\,\mathrm{tr}\!\left[\S(f_k)^{-1}\Y(f_k)\right]\right), \\
\mathcal{L}_{\mathrm{coarse}}(\S\mid\Y^{*})
&\propto \prod_{h=1}^{N_c}\left|\S(\bar f_h)\right|^{-N_bN_h}
\exp\!\left(-\frac{1}{T_b}\,\mathrm{tr}\!\left[\S(\bar f_h)^{-1}\Y_h^{*}\right]\right).
\end{align*}

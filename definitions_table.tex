\begin{table*}[t]
    \centering
    \small
    \renewcommand{\arraystretch}{1.15}
    \begin{tabular}{p{0.16\textwidth} p{0.78\textwidth}}
        \toprule
        \textbf{Symbol} & \textbf{Meaning} \\
        \midrule
        $T$ & Total observation duration. \\
        $\Delta_t$ & Sampling time interval; in the code this is typically denoted by \texttt{dt}, with $\Delta_t = \texttt{dt} = 1 / f_s$. \\
        $f_s$ & Sampling frequency, $f_s = 1 / \Delta_t$. \\
        $n$ & Number of time samples per channel, with $n = T / \Delta_t$. \\
        $p$ & Number of channels. \\
        $f_{Ny}$ & Nyquist frequency. \\
        $\Delta_f$ & Frequency resolution, $\Delta_f = 1 / (n \Delta_t) = 1 / T$. \\
        $f_k$ & Discrete Fourier frequency indexed by $k$, with $f_k = k \Delta_f = k / T$. \\
        $N$ & Full-series positive-frequency count used in the paper, after dropping the DC bin: $N = n/2$ if $n$ is even and $N = (n-1)/2$ if $n$ is odd. In the code, \texttt{fft\_data.N} instead means the \emph{current retained grid size} after any band truncation or coarse graining. \\
        $N_b$ & Number of non-overlapping time blocks. \\
        $L_b$ & Number of samples per block, $L_b = n / N_b$. \\
        $T_b$ & Block duration, $T_b = L_b \Delta_t = T / N_b$. \\
        $N_l$ & Number of retained positive block frequencies on the fine grid used by the blocked likelihood. With even $L_b$ and no additional band truncation, $N_l = L_b / 2 = n / (2N_b)$. In code diagnostics this retained fine-grid count is often reported as $N_{\ell}$. \\
        $N_c$ & Number of coarse-grain frequency bins. In the code this is \texttt{Nc}. \\
        $N_h$ & Number of fine-grid frequencies aggregated into each coarse-grain bin. In the code this is \texttt{Nh}. \\
        $\Z$ & Observed multivariate time series, $\Z = (\Z_1, \ldots, \Z_n)^{\intercal} \in \mathbb{R}^{n \times p}$. \\
        $\d(f_k)$ & Discrete Fourier transform vector at frequency $f_k$, $\d(f_k) = (d_1(f_k), d_2(f_k), \ldots, d_p(f_k))^{\intercal}$. \\
        $\S(f_k)$ & $p \times p$ Hermitian positive semidefinite spectral density matrix at frequency $f_k$. \\
        $\I^{(i)}(f_k)$ & Periodogram matrix for block $i$ at frequency $f_k$, $\I^{(i)}(f_k) = \d^{(i)}(f_k)\d^{(i)}(f_k)^*$. \\
        $\bar{\I}(f_k)$ & Block-averaged periodogram, $\bar{\I}(f_k) = \frac{1}{N_b}\sum_{i=1}^{N_b}\I^{(i)}(f_k)$. \\
        $\Y(f_k)$ & Summed multivariate periodogram / Wishart statistic, $\Y(f_k) = \sum_{i=1}^{N_b}\d^{(i)}(f_k)\d^{(i)}(f_k)^* = N_b \bar{\I}(f_k)$. \\
        $\U^{(k)}$ & Matrix of eigenvector-weighted Wishart factors at frequency $f_k$, with columns $\u_{\nu}^{(k)}$ such that $\Y(f_k) = \sum_{\nu=1}^{p}\u_{\nu}^{(k)}\u_{\nu}^{(k)*}$. This is the object represented in code by the complex matrix built from \texttt{u\_re} and \texttt{u\_im}. \\
        $J_h$ & The $h$th coarse-grain bin, a consecutive subset of retained frequencies. \\
        $\bar{f}_h$ & Representative frequency for coarse bin $J_h$. The current manuscript defines this as the midpoint Fourier frequency of $J_h$, which matches the implementation. \\
        $\Y_h^{*}$ & Coarse-grained Wishart statistic for bin $J_h$, $\Y_h^{*} = \sum_{f \in J_h}\Y(f)$. \\
        \bottomrule
    \end{tabular}
    \caption{Notation used in the multivariate Whittle likelihood, blocked periodogram construction, and equal-bin coarse graining.}
    \label{tab:definitions}
\end{table*}
